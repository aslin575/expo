import React, { useState } from 'react';
import { View, Text, TextInput, Button, StyleSheet } from 'react-native';
import { collection, addDoc } from 'firebase/firestore';
import { db } from './firebase';

export default function PostForm() {
  const [title, setTitle] = useState('');
  const [message, setMessage] = useState('');

  const handleSubmit = async () => {
    if (!title || !message) {
      alert("Tafadhali jaza kichwa na ujumbe.");
      return;
    }

    try {
      await addDoc(collection(db, 'posts'), {
        title,
        message,
        timestamp: new Date()
      });
      alert("Ujumbe umetumwa!");
      setTitle('');
      setMessage('');
    } catch (error) {
      console.error("Hitilafu:", error);
      alert("Kuna tatizo kutuma ujumbe.");
    }
  };

  return (
    <View style={styles.container}>
      <Text style={styles.label}>Kichwa cha Ujumbe</Text>
      <TextInput
        style={styles.input}
        placeholder="Mfano: Isaya 58"
        value={title}
        onChangeText={setTitle}
      />

      <Text style={styles.label}>Ujumbe</Text>
      <TextInput
        style={styles.textArea}
        placeholder="Andika ujumbe wako hapa..."
        multiline
        value={message}
        onChangeText={setMessage}
      />

      <Button title="Tuma Ujumbe" onPress={handleSubmit} />
    </View>
  );
}

const styles = StyleSheet.create({
  container: { padding: 20 },
  label: { fontWeight: 'bold', marginBottom: 5 },
  input: { borderWidth: 1, borderColor: '#ccc', marginBottom: 10, padding: 10, borderRadius: 5 },
  textArea: { borderWidth: 1, borderColor: '#ccc', height: 100, padding: 10, borderRadius: 5, marginBottom: 10 }
});
